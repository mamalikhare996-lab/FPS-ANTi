--// Small UI helpers
local function make(className, properties, parent)
    local instance = Instance.new(className)
    for key, value in pairs(properties or {}) do
        instance[key] = value
    end
    instance.Parent = parent
    return instance
end

local function addCorner(object, radius)
    make("UICorner", { CornerRadius = UDim.new(0, radius or 10) }, object)
end

local function addStroke(object, color, thickness, transparency)
    return make("UIStroke", {
        Color = color or COLORS.stroke,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
    }, object)
end

local function addGradient(object, first, second, rotation)
    return make("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, first),
            ColorSequenceKeypoint.new(1, second),
        }),
        Rotation = rotation or 0,
    }, object)
end

local function textLabel(parent, text, size, position, fontSize, color)
    return make("TextLabel", {
        BackgroundTransparency = 1,
        Text = text,
        Size = size,
        Position = position,
        Font = Enum.Font.GothamMedium,
        TextSize = fontSize or 14,
        TextColor3 = color or COLORS.white,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
    }, parent)
end

--// Generic drag behavior for mouse and touch
local function makeDraggable(handle, movingObject)
    local dragging = false
    local dragInput
    local dragStart
    local startPosition

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = movingObject.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            local delta = input.Position - dragStart
            movingObject.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)
end

local function showToast(message, isSuccess)
    toastSerial += 1

    toastText.Text = message
    toastText.TextColor3 = isSuccess and COLORS.green or COLORS.red
    TweenService:Create(toast, TweenInfo.new(0.22, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
        Position = UDim2.new(0.5, 0, 0, 18),
    }):Play()
    task.delay(2, function()
        if thisToast == toastSerial and toast.Parent then
            TweenService:Create(toast, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
                Position = UDim2.new(0.5, 0, 0, -60),
            }):Play()
        end
    end)

local function findPlayerByName(query)
    local normalized = string.lower(string.gsub(query, "^%s*(.-)%s*$", "%1"))
    if normalized == "" then return nil end
    for _, player in ipairs(Players:GetPlayers()) do
        if string.lower(player.Name) == normalized or string.lower(player.DisplayName) == normalized then
            return player
        end
    end
    return nil
end

local function selectPlayer(player)
    if not player or player == localPlayer then
        showToast("Invalid Target", false)
        return false
    end
    state.selectedUserId = player.UserId
    state.selectedName = player.Name
    nameBox.Text = player.Name
    selectedLabel.Text = "Selected: " .. player.DisplayName .. "  (@" .. player.Name .. ")"
    selectedLabel.TextColor3 = COLORS.green
    showToast("Target Selected", true)
    return true
end

selectButton.Activated:Connect(function()
    local target = findPlayerByName(nameBox.Text)
    if not target or target == localPlayer then
        state.selectedUserId = nil
        state.selectedName = ""
        selectedLabel.Text = "Selected: none"
        selectedLabel.TextColor3 = COLORS.muted
        showToast("Invalid Target", false)
        return
    end
    selectPlayer(target)
end)

local function rebuildPlayerList()
    for _, child in ipairs(userList:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end
    local others = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then table.insert(others, player) end
    end
    table.sort(others, function(a, b)
        return string.lower(a.Name) < string.lower(b.Name)
    end)
    for _, player in ipairs(others) do
        local row = make("TextButton", {
            Name = "User_" .. player.UserId,
            Size = UDim2.new(1, -2, 0, 30),
            BackgroundColor3 = COLORS.panel2,
            BorderSizePixel = 0,
            Text = player.DisplayName .. "   (@" .. player.Name .. ")",
            TextColor3 = COLORS.white,
            TextSize = 11,
            Font = Enum.Font.Gotham,
            TextXAlignment = Enum.TextXAlignment.Left,
            ZIndex = 5,
        }, userList)
        addCorner(row, 7)
        make("UIPadding", { PaddingLeft = UDim.new(0, 9) }, row)
        row.Activated:Connect(function() selectPlayer(player) end)
    end
    task.defer(function()
        userList.CanvasSize = UDim2.new(0, 0, 0, listLayout.AbsoluteContentSize.Y + 12)
    end)
end

local gradientConnection

local function setActive(enabled)
    state.active = enabled
    startText.Text = enabled and "ϟ   OFF" or "ϟ   START"
    if enabled then
        startText.TextColor3 = COLORS.cyan
        if gradientConnection then gradientConnection:Disconnect() end
        gradientConnection = RunService.RenderStepped:Connect(function(dt)
            if not startGradient.Parent then return end
            startGradient.Rotation = (startGradient.Rotation + dt * 100) % 360
        end)
    else
        startText.TextColor3 = COLORS.white
        if gradientConnection then
            gradientConnection:Disconnect()
            gradientConnection = nil
        end
        startGradient.Rotation = 0
    end
end
startButton.Activated:Connect(function()
    if not state.active and not state.selectedUserId then
        showToast("Select a target first", false)
        return
    end
    setActive(not state.active)
    showToast(state.active and "Demo Enabled" or "Demo Disabled", state.active)
end)

local function setSpeed(value)
    state.speed = math.clamp(math.floor(value + 0.5), 1, 100)
    local alpha = state.speed / 100
    sliderFill.Size = UDim2.new(alpha, 0, 1, 0)
    knob.Position = UDim2.new(alpha, 0, 0.5, 0)
    speedValue.Text = state.speed == 100 and "100 · MAX" or tostring(state.speed)
end
local sliding = false
local function updateSliderFromX(x)
    local width = sliderTrack.AbsoluteSize.X
    if width <= 0 then return end
    setSpeed(math.clamp((x - sliderTrack.AbsolutePosition.X) / width, 0, 1) * 100)
end
sliderTrack.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        sliding = true
        updateSliderFromX(input.Position.X)
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        updateSliderFromX(input.Position.X)
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        sliding = false
    end
end)
maxButton.Activated:Connect(function() setSpeed(100) end)
